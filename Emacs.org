        ;;; ------------------------------------------------------------
        ;;; Orderless
        ;;; ------------------------------------------------------------
        ;;; Orderless：
        ;;;   控制 Emacs completion 的匹配方式。
        ;;;
        ;;; 例如候选：
        ;;;
        ;;;   project-switch-to-buffer
        ;;;   project-current
        ;;;   project-find-file
        ;;;
        ;;; 输入：
        ;;;
        ;;;   pro buf
        ;;;
        ;;; 就可以分别匹配不同部分。
        ;;;
        ;;; 核心思想：
        ;;;   输入多个关键词，不要求它们连续出现。
        ;;; ------------------------------------------------------------

        (use-package orderless
          :demand t

          :config
          ;; 自定义一个 Orderless completion style。
          ;;
          ;; 支持：
          ;;   initialism
          ;;   literal
          ;;   regexp
          ;;
          ;; 例如：
          ;;   pfb
          ;;
          ;; 可以匹配：
          ;;   project-find-buffer
          ;;
          (orderless-define-completion-style orderless+initialism
            (orderless-matching-styles
             '(orderless-initialism
               orderless-literal
               orderless-regexp)))

          ;; 默认使用 Orderless。
          (setq completion-styles '(orderless)

                ;; 不使用 Emacs 默认的 category defaults。
                completion-category-defaults nil

                ;; 默认匹配方式：
                ;;   literal
                ;;   regexp
                orderless-matching-styles
                '(orderless-literal
                  orderless-regexp)

                ;; 文件路径使用 partial-completion。
                ;;
                ;; 例如：
                ;;   ~/pro/src
                ;;
                ;; 可以进行目录层级补全。
                completion-category-overrides
                '((file (styles partial-completion)))))

  #+begin_src emacs-lisp :tangle ~/.config/emacs/modules/dh-func.el :mkdirp yes
        (defun dh/duplicate-line ()
      "Duplicate current line and keep cursor column."
      (interactive)
      (let ((col (current-column))
            (line (thing-at-point 'line t)))
        (save-excursion
          (end-of-line)
          (insert "\n" line))
        (forward-line 1)
        (move-to-column col)))

    (global-set-key (kbd "C-,")
                    #'dh/duplicate-line)

    (use-package move-text
       :ensure t
       :bind
        (("M-<up>" . move-text-up)
        ("M-<down>" . move-text-down)))

    (provide 'dh-func)
